\chapter{Atajos de teclado}\label{cap:d-atajos}

Funcionan en casi todos los programas y escritorios.

\begin{center}
\begin{tabularx}{\textwidth}{lX}
\toprule
Teclas & Qué hacen \\
\midrule
\keys{\ctrl + C} / \keys{\ctrl + X} / \keys{\ctrl + V} & Copiar / cortar / pegar \\
\keys{\ctrl + Z} / \keys{\ctrl + Y} & Deshacer / rehacer \\
\keys{\ctrl + A} & Seleccionar todo \\
\keys{\ctrl + S} & Guardar \\
\keys{\ctrl + P} & Imprimir \\
\keys{\ctrl + F} & Buscar \\
\keys{\ctrl + +} / \keys{\ctrl + -} & Agrandar / achicar \\
\keys{Alt + Tab} & Pasar de una ventana a otra \\
\keys{Alt + F4} & Cerrar la ventana \\
\keys{F2} & Cambiar el nombre de un archivo \\
\keys{Supr} & Mandar a la papelera \\
\keys{ImprPant} & Captura de pantalla \\
\bottomrule
\end{tabularx}
\end{center}

\section*{En XFCE}
\begin{center}
\begin{tabularx}{\textwidth}{lX}
\toprule
\keys{Alt + F2} & Buscar y abrir un programa \\
\keys{\ctrl + Alt + T} & Abrir la terminal \\
\keys{\ctrl + Alt + D} & Mostrar el escritorio \\
\keys{\ctrl + Alt + Supr} & Bloquear la pantalla \\
\bottomrule
\end{tabularx}
\end{center}

\section*{En Firefox}
\begin{center}
\begin{tabularx}{\textwidth}{lX}
\toprule

\keys{\ctrl + L} & Ir a la barra de direcciones \\
\keys{\ctrl + D} & Guardar en marcadores \\

\keys{F11} & Pantalla completa \\
\bottomrule

\end{center}
